    <script>
        function checkQuiz() {
            let score = 0;
            const answers = {
                q1: "false",
                q2: "false",
                q3: "true",
                q4: "true",
                q5: "false",
                q6: "true",
                q7: "false",
                q8: "true",
                q9: "false",
                q10: "true"
            };

            for (let q in answers) {
                const userAns = document.querySelector(`input[name="${q}"]:checked`);
                if (userAns && userAns.value === answers[q]) score++;
            }

            document.getElementById("result").textContent = `You scored ${score} out of 10.`;

            const explanationText = `
                <p><strong>1.</strong> False – Not all FYP videos are reliable.</p>
                <p><strong>2.</strong> False – Friends may share videos without checking.</p>
                <p><strong>3.</strong> True – Scientific journals check information carefully.</p>
                <p><strong>4.</strong> True – Academic websites usually provide reliable info.</p>
                <p><strong>5.</strong> False – Social media news is not always checked.</p>
                <p><strong>6.</strong> True – Government sites give official information.</p>
                <p><strong>7.</strong> False – Viral content is not always made by experts.</p>
                <p><strong>8.</strong> True – Fact-checking stops misinformation.</p>
                <p><strong>9.</strong> False – Influencers can be biased or mistaken.</p>
                <p><strong>10.</strong> True – Checking many sources helps confirm accuracy.</p>`;

            document.getElementById("explanations").innerHTML = explanationText;
        }
    </script>
